-- SYNTAX
# Docker'ın eski komutu (hala çalışır, ancak yenisi önerilir)
docker run hello-world

# Docker'ın yeni komutu (modern kullanım)
docker container run hello-world

-- HELP
# Docker komutları hakkında yardım almak için:
docker search --help

-- SEARCH (Görüntü Arama ve Çekme)
# Docker Hub üzerinde belirli bir görüntüyü (image) aramak:
docker search nginx

# Docker Hub'dan bir görüntüyü indirmek:
docker pull nginx

-- CONTAINER (Konteyner İşlemleri)
# Konteynerler, belirli uygulamaları izole bir ortamda çalıştırmak için kullanılır.
# Nginx görüntüsünü indir ve çalıştır (80 numaralı porta bağla):
docker container run --publish 80:80 nginx

# Nginx'i arka planda (detached mode) çalıştır:
docker container run --publish 81:80 --detach nginx

# Konteynere özel bir isim vererek çalıştır:
docker container run --publish 82:80 --detach --name nginx_my1 nginx

docker container run --publish 83:80 --detach --name nginx_my3 nginx

# Konteyner durduğunda otomatik olarak silinsin:
docker container run --publish 84:80 --detach --name nginx_my4 --rm nginx

docker container run --publish 85:80 --detach --name nginx_my5 --rm nginx:latest

docker container run --publish 86:80 --detach --name nginx_my6 --rm nginx:1.14

# Apache HTTP Server çalıştır:
docker container run --publish 1111:80 --name apache_my1 httpd

# Curl ile bağlantı testi:
sudo apt-get install curl 
curl localhost:8888

-- LIST (Konteynerleri Listeleme)
# Çalışan konteynerleri listele:
docker ps

# Tüm konteynerleri (çalışmayanlar dahil) listele:
docker container ls -a

-- STOP (Konteyner Durdurma)
# Bir konteyneri durdur:
docker container stop <CONTAINER_ID>
docker container stop <CONTAINER_NAME>

# Birden fazla konteyneri durdur:
docker container stop 568e5cf521c8 3c9b352b32e5 2df803256987

-- START (Konteyner Başlatma)
# Yeni bir konteyner çalıştır:
docker container run -p 90:80 -d --name nginx_my8 nginx

# Çalışan konteynerleri listele:
docker ps

docker container stop a369ec5420a5

# Önceden oluşturulmuş bir konteyneri başlat:
docker container start nginx_my8

docker ps

-- DELETE (Konteyner Silme)
# Tüm konteynerleri listele:
docker container ls -a

# Önce durdur, sonra sil:
docker container stop <CONTAINER_ID>
docker container rm <CONTAINER_ID>
docker container rm <CONTAINER_NAME>

# Birden fazla konteyneri sil:
docker container rm fd79bf97f41f 9ec0c0792b8e 83d7a4bb6721

-- DELETE FORCE (Zorla Silme)
# Çalışan bir konteyneri zorla sil:
docker container rm -f <CONTAINER_ID>
docker container rm -f <CONTAINER_NAME>

-- PRUNE DELETE (Tüm Kullanılmayan Konteynerleri Temizleme)
docker container prune

---------------------------------------------------------------
-- ERROR (Hata Senaryoları)
1- Çalışan bir konteyneri doğrudan silmeye çalışırsanız:
docker container rm 267a14209476
HATA: "container is running" hatası alırsınız. Önce durdurmanız veya -f (force) kullanmanız gerekir.

2- Aynı isimde konteyner oluşturmaya çalışırsanız:
HATA: "Conflict. The container name \"/nginx_my1\" is already in use" hatası alırsınız. 
Çözüm: Konteyneri yeniden adlandırın veya silin.

3- Aynı porta sahip iki konteyner oluşturmaya çalışırsanız:
HATA: "Bind for 0.0.0.0:82 failed: port is already allocated" hatası alırsınız. 
Çözüm: Farklı bir port kullanın veya mevcut portu kullanan konteyneri durdurun.

-- DOCKER KOMUTLARI (Yardım İçin)
docker container --help

# Konteynerler üzerinde yapılabilecek işlemler:
Commands:
  attach      Bir konteynere bağlan
  commit      Konteyner değişikliklerinden yeni bir görüntü oluştur
  cp          Konteyner ile yerel dosya sistemi arasında dosya kopyala
  create      Yeni bir konteyner oluştur
  diff        Konteyner dosya sistemindeki değişiklikleri gör
  exec        Çalışan bir konteyner içinde komut çalıştır
  export      Konteyner dosya sistemini bir tar arşivi olarak dışa aktar
  inspect     Konteyner hakkında detaylı bilgi göster
  kill        Çalışan bir konteyneri zorla sonlandır
  logs        Konteyner loglarını göster
  ls          Konteynerleri listele
  pause       Konteyner içindeki tüm işlemleri duraklat
  port        Port eşleştirmelerini listele
  prune       Tüm durdurulmuş konteynerleri temizle
  rename      Konteyneri yeniden adlandır
  restart     Konteyneri yeniden başlat
  rm          Bir veya daha fazla konteyneri sil
  run         Yeni bir konteyner oluştur ve çalıştır
  start       Durdurulmuş bir konteyneri başlat
  stats       Konteyner kaynak kullanım istatistiklerini göster
  stop        Bir veya daha fazla çalışan konteyneri durdur
  top         Konteyner içinde çalışan işlemleri göster
  unpause     Durdurulmuş bir konteyneri devam ettir
  update      Bir veya daha fazla konteynerin konfigürasyonunu güncelle
  wait        Konteyner durana kadar bekle ve çıkış kodunu göster

# Daha fazla detay için:
Run 'docker container COMMAND --help'
